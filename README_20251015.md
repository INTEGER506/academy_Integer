## ■ 성적관리

###  성적 창출 방법 재정의
    - 폼에서 중간, 기말, 과제, 출석 점수를 작성
    - 총점 산출 (서버 계산) [ex) 중간 30%, 기말 40%, 과제 20%, 출석 10%]
        ㄴ 위의 비율을 고정으로 할지 변경가능하게[폼제작?] 할지 결정
    - 산출된 총점으로 순위를 매겨 비율에 따라 점수(A+[10%],A[10%],B[10%]) 환산
        ㄴ 위의 비율을 고정으로 할지 변경가능하게[폼제작?] 할지 결정
    - 점수(A+)를 취득학점(4.5)으로 환산  [db에서 score,gpa에 들어감]

    -> 결론
        - 관리자 : Alphabet(점수) 비율 관리 가능        [폼 필요]
                    ㄴ 정의된 비율은 교수가 조회가능
        - 교수   :  중간,기말,과제,출석 비율 작성 관리 가능 [폼 필요]
        - 정리   : 관리자 Alphabet 비율 관리 -> 교수 점수 가중치 관리
                      -> 점수(중간,기말,과제,출석) 작성 시 작성
                      -> 총점계산 -> Alphabet비율에 따른 자동 변환
                      -> Alphabet의 자동 취득학점(Score)변환
                      -> 학생 별 취득학점, 평균학점(gpa) 계산

# 스키마 재 정의
-- 성적은 1학기에 한번 등록 / 총 1년에 2번
-- 성적 / grade

    CREATE TABLE grade(
    id 		NUMBER 		PRIMARY KEY,
    alphabet 	VARCHAR2(3),		-- 점수 (A+,A,~F)
    GPA 		NUMBER(2,1),		-- 평균 학점 (3.5)
    score 	NUMBER(2,1),		-- 취득 학점 (3.5)
    total_int 	NUMBER(3),		-- 총점(자연수)(예: 90, 80)

	--교수 입력 점수
	mid_exam	NUMBER(3),				--중간 고사
	final_exam	NUMBER(3),				--기말 고사
	assignment NUMBER(3),				--과제
	attendance NUMBER(3),				--출석

	student_id  NUMBER,
	CONSTRAINT FK_grade_student FOREIGN KEY (student_id) REFERENCES student(id),
	enrollment_id NUMBER,
	CONSTRAINT FK_grade_enrollment FOREIGN KEY (enrollment_id) REFERENCES enrollment(id)
    );

## 교수용 과목 점수 가중치 테이블
    -- 중간(%), 기말(%), 과제(%), 출석(%)
    CREATE TABLE grade_system (
    id				NUMBER		PRIMARY KEY,
    mid_exam_ratio		NUMBER(5,2)	DEFAULT 30 NOT NULL,		-- 중간 % , 기본값 30%
    final_exam_ratio		NUMBER(5,2)	DEFAULT 40 NOT NULL,		-- 중간 % , 기본값 30%
    assignment_ratio		NUMBER(5,2)	DEFAULT 20 NOT NULL,		-- 중간 % , 기본값 30%
    attendance_ratio		NUMBER(5,2)	DEFAULT 10 NOT NULL,		-- 중간 % , 기본값 30%
    course_id			NUMBER		NOT NULL,					-- 과목
    CONSTRAINT FK_gs_course FOREIGN KEY (course_id) REFERENCES course(id)
    );

## 관리자 Alphabet 비율 테이블
    CREATE TABLE alphabet_system (
    id			NUMBER		PRIMARY KEY,
    course_id		NUMBER,							-- Null이면 글로벌, 값이 있으면 해당 강의 전용
    alphabet		VARCHAR2(3),						-- 점수 (A+,A,~F)
    boundary		NUMBER(5,2),						-- 점수별 경계값 (0~100%) - 퍼센트 기반
    description		VARCHAR2(100)						-- 설명 (선택사항)
    );
## 시퀀스
    CREATE SEQUENCE grade_seq START WITH 1 INCREMENT BY 1;			-- 성적관리
    CREATE SEQUENCE grade_system_seq START WITH 1 INCREMENT BY 1;	-- 교수 점수 가중치
    CREATE SEQUENCE alphabet_system_seq START WITH 1 INCREMENT BY 1;	-- 관리자 Alphabet 비율
